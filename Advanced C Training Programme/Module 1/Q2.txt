#include <stdio.h>
#include <stdlib.h>

void rearrange(int *arr, int n) {
    int *temp = (int *)malloc(n * sizeof(int));
    int *ptr = temp;

    for (int *p = arr; p < arr + n; p++)
        if (*p % 2 == 0)
            *ptr++ = *p;

    for (int *p = arr; p < arr + n; p++)
        if (*p % 2 != 0)
            *ptr++ = *p;

    for (int i = 0; i < n; i++)
        *(arr + i) = *(temp + i);

    free(temp);
}

int main() {
    int a[100], n;
    printf("Enter array size: ");
    scanf("%d", &n);
    printf("Enter array elements:\n");
    for (int i = 0; i < n; i++)
        scanf("%d", a + i);

    rearrange(a, n);

    printf("Rearranged array:\n");
    for (int i = 0; i < n; i++)
        printf("%d ", *(a + i));
    printf("\n");

    return 0;
}
