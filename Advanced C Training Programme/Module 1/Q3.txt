#include <stdio.h>

int search(int arr[100][100], int n, int key) {
    int i = 0, j = n - 1;
    while (i < n && j >= 0) {
        if (arr[i][j] == key)
            return 1;
        else if (arr[i][j] > key)
            j--;
        else
            i++;
    }
    return 0;
}

int main() {
    int n, x;
    int mat[100][100];

    printf("Enter matrix size (n): ");
    scanf("%d", &n);

    printf("Enter matrix elements:\n");
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &mat[i][j]);

    printf("Enter key to search: ");
    scanf("%d", &x);

    if (search(mat, n, x))
        printf("Key found in matrix.\n");
    else
        printf("Key not found.\n");

    return 0;
}
